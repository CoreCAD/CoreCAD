# that cmake is allowed to overlay onto the build tree. Files not matching
# any pattern are ignored. Lines starting with # are comments.

# The source tree is never modified — overrides only exist in the build tree.
#
# Note: patterns without a matching corecad_stage_resources() call in cmake
# are treated as documentation only (e.g. installer icons that already live
# in the source tree and don't need build-tree staging).
